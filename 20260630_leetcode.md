57. Insert Interval


class Solution {
public:
    vector<vector<int>> insert(vector<vector<int>>& intervals, vector<int>& newInterval) {
        vector<vector<int>> result;
        int n = intervals.size();
        int i = 0;
        
        // ==========================================
        // 【第一階段：左側安全區】(無交集，直接放入)
        // ==========================================
        // 條件：只要「目前區間的右邊界」嚴格小於「新區間的左邊界」
        // 代表這個區間完全在新區間的左邊，絕對不可能重疊。
        while (i < n && intervals[i][1] < newInterval[0]) { 
            result.push_back(intervals[i]);
            i++;
        }
        
        // ==========================================
        // 【第二階段：戰區 / 融合區】(有交集，不斷擴張新區間)
        // ==========================================
        // 條件：只要「目前區間的左邊界」小於等於「新區間的右邊界」
        // 因為經過了第一階段，能走到這裡的區間，其右邊界必然 >= newInterval[0]。
        // 加上這裡的左邊界 <= newInterval[1] 條件，數學上保證了兩者「絕對有交集」。
        while (i < n && intervals[i][0] <= newInterval[1]) { 
            // 發生重疊，不急著放進 result，而是讓 newInterval 把這個區間「吃掉」
            // 融合後的左邊界：取兩者左邊界的極小值
            newInterval[0] = min(intervals[i][0], newInterval[0]);
            // 融合後的右邊界：取兩者右邊界的極大值
            newInterval[1] = max(intervals[i][1], newInterval[1]);
            i++;
        }
        // 戰區結束，新區間已經吸收了所有跟它重疊的區間，此時才能把它正式加入結果中
        result.push_back(newInterval);

        // ==========================================
        // 【第三階段：右側安全區】(無交集，直接放入)
        // ==========================================
        // 剩下的所有區間，其左邊界一定都嚴格大於新區間的右邊界。
        // 所以直接把它們全部照抄進 result 即可。
        while (i < n && intervals[i][0] > newInterval[1]) { 
            result.push_back(intervals[i]);
            i++;
        }
        
        return result;
    }
};
