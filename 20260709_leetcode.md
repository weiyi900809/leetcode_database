61. Rotate List

/**
 * Definition for singly-linked list.
 * struct ListNode {
 * int val;
 * ListNode *next;
 * ListNode() : val(0), next(nullptr) {}
 * ListNode(int x) : val(x), next(nullptr) {}
 * ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* rotateRight(ListNode* head, int k) {
        
        // 🛡️ 【防禦機制】
        // 如果 list 是空的，或是只有一個節點，無論怎麼旋轉都不會改變，直接回傳即可。
        if(head == nullptr || head->next == nullptr ) return head; 
        
        // 📦 【配置虛擬頭節點】
        // dummy.next 永遠指向「當前 List 的真正起點」，方便最後重組時直接取用。
        ListNode dummy(0, head);
        
        int list_length = 0;
        // 📍 node1：探路者，負責走到原始 List 的尾巴 (舊 tail)，同時計算總長度
        ListNode* node1 = &dummy; 
        
        // ==========================================
        // 🔄 【Round 1：計算長度並鎖定舊尾巴】
        // ==========================================
        while(node1 != nullptr && node1->next != nullptr){ 
            list_length++;
            node1 = node1->next;
        }
        // 此時，node1 已經停在原本 List 的最後一個節點 (舊 tail)。
        
        // ==========================================
        // 🧮 【Round 2：數學精簡與邊界判斷】
        // ==========================================
        // 當 k 大於等於 list_length 時，旋轉一整圈等於沒轉，所以直接取餘數。
        k = k % list_length;
        
        // 如果餘數為 0，代表旋轉後跟原本一模一樣，直接提早結束，節省效能。
        if(k == 0) return head;
        
        // ==========================================
        // ✂️ 【Round 3：尋找斷點並重新接合】
        // ==========================================
        // 尋找新的 tail：它會落在原本正向數過來的第 (長度 - k) 個節點。
        // 因為我們要站在它的前一格來操作 next，所以步數是 list_length - k - 1。
        int count = list_length - k - 1; 
        
        // 📍 node2：定位者，負責走到「旋轉後的新尾巴 (新 tail)」
        ListNode* node2 = head; 
        while(count > 0){
            node2 = node2->next;
            count--;
        }
        // 此時，node2 停在新的 tail，而 node2->next 就是新的 head。

        // 🔗 【實施指針重組：連環與切斷】
        // 1. 首尾相連：將舊的 tail 接上舊的 head (形成一個完整的環)
        node1->next = dummy.next; 
        
        // 2. 確立新起點：將 dummy 指向新的 head (讓 dummy.next 永遠是正確答案)
        dummy.next = node2->next; 
        
        // 3. 斷開環節：將新的 tail 的 next 設為 nullptr，確立新的 List 終點
        node2->next = nullptr; 
        
        return dummy.next;
    }
};
