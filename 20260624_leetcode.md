167. Two Sum II - Input Array Is Sorted



class Solution {
public:
    vector<int> twoSum(vector<int>& numbers, int target) {
        // 1. 【雙指標初始化】
        // 左手指標指著最小的數字 (最左邊)
        // 右手指標指著最大的數字 (最右邊)
        int left = 0;
        int right = numbers.size() - 1;
        
        // 2. 【雙向逼近邏輯】
        // 只要左右手還沒交會，就繼續找 (題目保證只有唯一解，且不能重複使用同一個元素)
        while (left < right) {
            
            // 算出目前的總和
            int add_result = numbers[left] + numbers[right];
            
            if (add_result == target) {
                // 🎯 命中紅心！
                // 題目要求回傳「1-indexed」(從 1 開始數) 的陣列索引，所以都要 +1
                // 這裡使用現代 C++ 語法，直接建構並回傳 vector
                return {left + 1, right + 1};
            }
            else if (add_result > target) {
                // 📉 總和太大了！
                // 因為陣列已經「從小到大排序」好，要讓總和變小，
                // 唯一的選擇就是把「右手 (比較大的數字)」往左邊移一點。
                right--;
            }
            else {
                // 📈 總和太小了！
                // 唯一的選擇就是把「左手 (比較小的數字)」往右邊移一點。
                left++;
            }
        }
        
        // 理論上程式不會走到這裡，因為題目保證「一定有解」
        return {}; 
    }
};
