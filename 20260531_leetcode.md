605. Can Place Flowers


class Solution {
public:
    bool canPlaceFlowers(vector<int>& flowerbed, int n) {
        int counter = n;

        // 【極端測資防禦】如果目標是種 0 朵花，代表不需要種，直接回傳 true (成功)
        if(n == 0) return true;
        
        for(int i = 0; i < flowerbed.size(); i++){
            
            // 條件 1：首先確保自己腳下這個位置是空的，才有資格繼續檢查左右
            if(flowerbed[i] == 0 ){
                
                // 條件 2：【左側防護盾】
                // 如果自己在最左邊 (i==0)，左邊視為安全；否則檢查左邊那格是不是空的。
                // 這裡利用了 || 的短路特性，只要 i==0 成立，就不會執行 flowerbed[i-1]，完美避開越界！
                if(i == 0 || flowerbed[i-1] == 0 ){
                    
                    // 條件 3：【右側防護盾】
                    // 如果自己在最右邊 (i==size-1)，右邊視為安全；否則檢查右邊那格是不是空的。
                    if(i == flowerbed.size()-1 || flowerbed[i+1] == 0 ){
                        
                        // 三關全過，成功種下一朵花！
                        // 【關鍵狀態更新】必須把這裡標記為 1，否則下一圈檢查左邊時會誤判
                        flowerbed[i] = 1; 
                        counter--;
                        
                        // 【效能優化：提前煞車】
                        // 只要達到目標數量，就不需要再浪費時間掃描後面的陣列了，直接收工
                        if(counter <= 0){
                            return true;
                        }
                    }
                }
            }
        }
        
        // 整個陣列都掃完了，還是沒達到目標數量，宣告失敗
        return false;
    }
};
