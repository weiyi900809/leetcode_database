15. 3Sum

class Solution {
public:
    vector<vector<int>> threeSum(vector<int>& nums) {
        // 宣告用來裝載最終答案的二維陣列
        vector<vector<int>> result;
        
        // 1. 第一步：陣列排序 (Sorting)
        // 【架構意義】系統降維打擊的基礎。沒有排序，雙指標就沒有「方向感」，也無法輕易揪出重複值。
        sort(nums.begin(), nums.end());

        // 2. 走訪陣列，將 nums[i] 視為每次配對時「固定不動的第一個數字」
        for(int i = 0; i < nums.size(); i++) {
            
            // 🛑 【效能提早煞車】
            // 因為陣列已經從小到大排序，如果最小的數字都大於 0，
            // 後面三個正數相加絕不可能等於 0，直接提早結束整個系統運算。
            if(nums[i] > 0){
                break;
            }
            
            // ⏭️ 【跳過重複機制 1：固定點排雷】
            // 確保 i > 0 防止陣列越界。
            // 如果現在固定的數字跟上一回合一樣，代表這個數字能湊出的組合剛剛都找過了，直接跳過。
            if(i > 0 && nums[i] == nums[i-1]){
                continue;
            }
            
            // 3. 設定雙指標的初始「位置 (Index)」
            // left 負責尋找較小的數，從 i 的下一格開始 (不走回頭路)
            // right 負責尋找較大的數，從陣列最後一格開始 (建立最大搜捕網)
            int left = i + 1;
            int right = nums.size() - 1;
            
            // 4. 啟動雙指標相向滑動尋寶 (必須確保左邊永遠在右邊的左側，防止交叉撞車)
            while(left < right){
                // 計算當下三個位置的「數值」總和
                int sum = nums[i] + nums[left] + nums[right];
                
                if (sum == 0){
                    // 🎉 完美命中目標！打包收錄進結果清單
                    result.push_back({nums[i], nums[left], nums[right]});
                    
                    // ⏭️ 【跳過重複機制 2：雙指標排雷】
                    // 記憶當下正確配對的數值，準備跳過後續所有一模一樣的廢物資料
                    int now_left = left;
                    int now_right = right;
                    
                    // 讓 left 一直往右推，直到踩到一個「跟剛剛不一樣」的新數字為止
                    while(left < right){
                        left++;
                        if(nums[left] != nums[now_left]) break;
                    }
                    
                    // 讓 right 一直往左推，直到踩到一個「跟剛剛不一樣」的新數字為止
                    while(left < right){
                        right--;
                        if(nums[right] != nums[now_right]) break;
                    }
                }
                else if (sum < 0){
                    // 總和太小，需要更大的數字來補，所以左指標往右移
                    left++;
                }
                else { // sum > 0
                    // 總和太大，需要更小的數字來平衡，所以右指標往左移
                    right--;
                }
            }
        }
        
        // 5. 將收集完畢的結果回傳給系統
        return result;
    }
};
