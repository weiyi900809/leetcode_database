103. Binary Tree Zigzag Level Order Traversal

/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<vector<int>> zigzagLevelOrder(TreeNode* root) {
        
        // 🛡️ 【防護網：空樹檢查】
        if(root == nullptr) return {};
        
        vector<vector<int>> result;
        
        // 📦 【基礎設施：BFS 排隊隊伍】
        queue<TreeNode*> node_list;
        node_list.push(root);
        
        // 📸 【快門機制：鎖定當前層的總人數】
        int queue_count = node_list.size();
        
        // 📍 【狀態旗標：控制輸出方向】
        // false (0) 表示正常順序：從左到右
        // true  (1) 表示反轉順序：從右到左
        bool direction = false; 
        
        // ==========================================
        // 🔄 【外層迴圈：垂直向下切換層級】
        // ==========================================
        while(!node_list.empty()){
            
            // temp 用來暫存「當前這層」的所有數字
            vector<int> temp;
            
            // ==========================================
            // 🔄 【內層迴圈：穩定的水平走訪 (永遠由左至右)】
            // ==========================================
            for(int i = 0; i < queue_count; i++){
                
                TreeNode* curr = node_list.front();
                
                // 1. 繁衍下一代：結構走訪"絕對不變"，永遠先放左小孩，再放右小孩
                // 這保證了 Queue 裡面的節點永遠是乾淨、正規的由左至右排列
                if(curr->left != nullptr){
                    node_list.push(curr->left);
                }
                if(curr->right != nullptr){
                    node_list.push(curr->right);
                }
                
                // 2. 收集當前節點的值
                temp.push_back(curr->val);
                
                // 3. 處理完畢，離開隊伍
                node_list.pop();
            }
            
            // ==========================================
            // 🎯 【加工與收割：輸出邏輯脫鉤】
            // ==========================================
            // 如果當前層的旗標指示「需要從右到左」，我們直接將這層的陣列反轉即可。
            // 這樣完全不會污染到 Queue 裡下一層節點的物理順序！
            if(direction == true) {
                reverse(temp.begin(), temp.end());
            }
            
            // 將加工好的這層結果，正式裝進總結果中
            result.push_back(temp);
            
            // 🔄 狀態更新：切換方向旗標 (false 變 true, true 變 false)
            direction = !direction;
            
            // 📸 再次按下快門，更新下一層的總人數
            queue_count = node_list.size();
        }
        
        return result;
    }
};
