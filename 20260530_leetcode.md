1290. Convert Binary Number in a Linked List to Integer


/**
 * Definition for singly-linked list.
 * struct ListNode {
 * int val;
 * ListNode *next;
 * ListNode() : val(0), next(nullptr) {}
 * ListNode(int x) : val(x), next(nullptr) {}
 * ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    int getDecimalValue(ListNode* head) {
        // result 用來累積最終的十進位數值
        int result = 0;
        
        // 使用 curr 指標來走訪鏈結串列，避免直接修改 head 導致丟失源頭
        ListNode* curr = head;
        
        // 從最高位 (Most Significant Bit) 一路走到最低位
        while(curr != NULL){
            // 【核心魔法：位元左移】
            // 每當我們讀到下一個位元時，代表前面累積的所有數字，權重都自動升級了 (乘以 2)
            // 使用左移運算子 <<= 1 (等同於 result = result * 2)，在 CPU 執行上比乘法更快
            result <<= 1;
            
            // 把當前節點的值 (0 或 1) 加到最低位
            // 這裡也可以寫成 result = result | curr->val; (使用位元 OR 運算，速度一樣快)
            result += curr->val;
            
            // 指標推進，前往下一個節點
            curr = curr->next;
        }
        
        return result;
    }
};
