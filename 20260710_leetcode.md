86. Partition List

/**
 * Definition for singly-linked list.
 * struct ListNode {
 * int val;
 * ListNode *next;
 * ListNode() : val(0), next(nullptr) {}
 * ListNode(int x) : val(x), next(nullptr) {}
 * ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    ListNode* partition(ListNode* head, int x) {
        
        // 📦 【基礎設施準備：雙虛擬節點】
        // 宣告在 Stack 上，用來當作兩個「新籃子」的起點。
        // dummy_less: 專門收集小於 x 的節點
        // dummy_greater: 專門收集大於等於 x 的節點
        ListNode dummy_less(0);
        ListNode dummy_greater(0);
        
        // 📍 【操作指標宣告】
        ListNode* node1 = head;           // 探路者：用來走訪並檢查原始 List 的每一個節點
        ListNode* node2 = &dummy_less;    // 收尾者 A：永遠指向 less_list 的最後一個節點 (tail)
        ListNode* node3 = &dummy_greater; // 收尾者 B：永遠指向 greater_list 的最後一個節點 (tail)

        // ==========================================
        // 🔄 【Round 1：物理分流 (Decoupling)】
        // ==========================================
        while(node1 != nullptr){
            
            if(node1->val < x){
                // 狀況 A：數值小於 x，裝進 less 籃子
                node2->next = node1;  // less 的尾巴牽起這個新節點
                node2 = node2->next;  // less 的尾巴指標往後推一格，確立新的 tail
            }
            else{
                // 狀況 B：數值大於等於 x，裝進 greater 籃子
                node3->next = node1;  // greater 的尾巴牽起這個新節點
                node3 = node3->next;  // greater 的尾巴指標往後推一格，確立新的 tail
            }
            
            // 探路者繼續往下一個節點前進
            node1 = node1->next;
        }

        // ==========================================
        // 🔗 【Round 2：防禦與重組 (Reassembling)】
        // ==========================================
        
        // ✂️ 【極度關鍵的防呆機制：斷開殘留指標】
        // 因為我們是「直接拿原始節點來串」，greater_list 的最後一個節點，
        // 它的 next 很可能還偷偷指著原始 List 中的某個小於 x 的節點。
        // 如果不斷開，重組後就會形成無窮迴圈 (Cycle)！
        node3->next = nullptr; 
        
        // 🔗 【雙軌合一】
        // 將 less_list 的尾巴，接上 greater_list 的真正開頭 (跳過 dummy)
        node2->next = dummy_greater.next;
        
        // 回傳 less_list 的真正開頭 (跳過 dummy)，這就是整串重新排列好的 List 起點
        return dummy_less.next;
    }
};
