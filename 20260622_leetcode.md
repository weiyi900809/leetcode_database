151. Reverse Words in a String

class Solution {
public:
    string reverseWords(string s) {
        
        // ==========================================
        // 【第一步：雙指標原地壓縮空格】
        // 目標：移除頭尾空格，並將單字間的多餘空格壓縮成一個
        // ==========================================
        int slow = 0;
        int fast = 0;
        
        for (fast = 0; fast < s.size(); fast++) {
            
            // 只要遇到不是空白的字元，就代表「抓到一個新單字的開頭了！」
            if (s[fast] != ' ') {
                
                // 防線：如果這不是整個字串的第一個單字，
                // 在開始寫入這個新單字之前，必須先手動補上一個空格。
                if (slow != 0) {
                    s[slow++] = ' ';
                }
                
                // 核心動作：一口氣搬移整個單字！
                // 只要 fast 還沒越界，且指向的還是字母 (非空格)，就繼續複製
                while (fast < s.size() && s[fast] != ' ') {
                    s[slow++] = s[fast++];
                }
                // 注意：當此 while 結束時，fast 剛好停在單字後面的第一個空格上。
            }
        }
        
        // 壓縮完畢，裁切掉 slow 後面殘留的舊字元，得到真正有效的乾淨字串
        s.resize(slow);
        
        // ==========================================
        // 【第二步：整體翻轉】
        // ==========================================
        // 物理效果："the sky is blue" -> "eulb si yks eht"
        reverse(s.begin(), s.end());

        // ==========================================
        // 【第三步：各別單字翻轉】
        // ==========================================
        slow = 0; // 重新利用 slow 當作每個單字的「起點」
        
        // 這裡的條件必須是 fast <= s.size()，故意讓 fast 走出陣列外一格
        for (fast = 0; fast <= s.size(); fast++) {
            
            // 觸發翻轉的兩個時機：
            // 1. fast == s.size()：走到了字串盡頭 (處理最後一個單字)
            // 2. s[fast] == ' '：遇到了空格 (處理中間的單字)
            if (fast == s.size() || s[fast] == ' ') {
                
                // 執行翻轉：區間為左閉右開 [slow, fast)
                reverse(s.begin() + slow, s.begin() + fast);
                
                // 將下一個單字的起點，設定在目前這個空格的下一格
                slow = fast + 1;
            }
        }
        
        return s; 
    }
};
