117. Populating Next Right Pointers in Each Node II

/*
// Definition for a Node.
class Node {
public:
    int val;
    Node* left;
    Node* right;
    Node* next;

    Node() : val(0), left(NULL), right(NULL), next(NULL) {}
    Node(int _val) : val(_val), left(NULL), right(NULL), next(NULL) {}
    Node(int _val, Node* _left, Node* _right, Node* _next)
        : val(_val), left(_left), right(_right), next(_next) {}
};
*/

class Solution {
public:
    Node* connect(Node* root) {
        // 📦 【基礎設施：下一層的虛擬起點】
        // dummy 用來當作「下一層」Linked List 的開頭。
        // 它的 next 永遠會指向「下一層的第一個真實節點」。
        Node* dummy = new Node();
        dummy->next = root; // 初始狀態：把 root 當作第一層，掛在 dummy 後面等待處理

        Node* curr;       // 📍 探路者：負責在「當前這層」由左向右走訪
        Node* dummy_tail; // 📍 串接者：負責在「下一層」不斷把子節點串起來的尾巴指標

        // ==========================================
        // 🔄 【外層迴圈：垂直向下切換層級】
        // ==========================================
        // 如果 dummy->next 是空的，代表上一層完全沒有留下任何子節點，整棵樹走訪結束
        while(dummy->next != nullptr){
            
            // 1. 確立當前層的起點 (把上一層串好的結果拿來用)
            curr = dummy->next; 
            
            // 2. 🎯 【極度關鍵的狀態重置 (State Reset)】
            // 進入新的一層後，必須把 dummy->next 清空。
            // 因為 dummy 馬上要被用來去收集「下下一層」的新節點。
            // 如果不清空，dummy->next 會一直卡在舊的節點，導致無窮迴圈或結構錯亂。
            dummy->next = nullptr; 
            
            // 3. 尾巴指標歸位，準備開始從 dummy 後面串接全新的下一層
            dummy_tail = dummy;

            // ==========================================
            // 🔄 【內層迴圈：水平向右走訪當前層】
            // ==========================================
            while(curr != nullptr){ 
                
                // 狀況 A：如果當前節點有左小孩，把它掛到下一層的尾巴上
                if(curr->left != nullptr){
                    dummy_tail->next = curr->left;
                    dummy_tail = dummy_tail->next; // 尾巴往後推進
                }
                
                // 狀況 B：如果當前節點有右小孩，同樣掛到下一層的尾巴上
                if(curr->right != nullptr){
                    dummy_tail->next = curr->right;
                    dummy_tail = dummy_tail->next; // 尾巴往後推進
                }
                
                // 當前節點的小孩都處理完了，透過「已經被上一輪建立好的 next 指標」，
                // 直接移動到當前層的下一個節點，這就是省下 Queue 的魔法！
                curr = curr->next;
            }
        }
        
        delete dummy;
        return root;
    }
};
