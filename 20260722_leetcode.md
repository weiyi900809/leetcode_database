199. Binary Tree Right Side View


/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    vector<int> rightSideView(TreeNode* root) {
        
        vector<int> result; // 用來存放從右邊看過去的所有數字
        
        // 🛡️ 【防護網：空樹檢查】
        // 如果樹是空的，直接回傳空陣列，這也是遞迴或迴圈起手式的良好習慣。
        if(root == nullptr) return {};
        
        // 📦 【基礎設施：BFS 排隊隊伍】
        // nodelist 扮演「等待處理的節點隊伍」
        queue<TreeNode*> nodelist;
        nodelist.push(root); // 把 Root 當作第一層的唯一節點，推入隊伍
        
        // 📸 【核心技巧：拍下當前層的快門】
        // 記錄「當前這層」總共有幾個人。
        int queue_count = nodelist.size(); 
        
        // ==========================================
        // 🔄 【外層迴圈：垂直向下切換層級】
        // ==========================================
        while(!nodelist.empty()){

            // ==========================================
            // 🔄 【內層迴圈：水平處理「當前這層」的所有人】
            // ==========================================
            // 嚴格限制只處理 queue_count 這麼多次，避免掃到「剛被加入隊伍的下一層節點」
            for(int i = 0; i < queue_count; i++){

                // 1. 叫隊伍最前面的人出來
                TreeNode* curr = nodelist.front();
                
                // 2. 繁衍下一代：如果他有左或右小孩，就叫小孩去隊伍的最尾端排隊 (屬於下一層)
                // ⚠️ 注意：必須先放左邊、再放右邊，這樣每一層的最後一個才會是最右邊的節點！
                if(curr->left != nullptr){
                    nodelist.push(curr->left);
                }
                if(curr->right != nullptr){
                    nodelist.push(curr->right);
                }
                
                // 🎯 【捕獲目標：誰是最右邊的人？】
                // 因為我們是從左到右把節點放進 Queue，
                // 所以當 i 走到當前層的最後一個 (queue_count - 1) 時，
                // 這個 curr 就是當前這層「最右邊的節點」！
                if(i == queue_count - 1){
                    result.push_back(curr->val);
                }
                
                // 3. 這個人處理完了，請他離開隊伍
                nodelist.pop();
            }
            
            // 📸 當這層的人都處理完離開後，現在 Queue 裡面剩下的，全部都是「下一層」的人了。
            // 再次按下快門，更新下一層的總人數，準備進入下一次外層迴圈。
            queue_count = nodelist.size();
        }
        
        // 回傳收集好的右側視圖
        return result;
    }
};
