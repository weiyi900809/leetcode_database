138. Copy List with Random Pointer

/*
// Definition for a Node.
class Node {
public:
    int val;
    Node* next;
    Node* random;
    
    Node(int _val) {
        val = _val;
        next = NULL;
        random = NULL;
    }
};
*/

class Solution {
public:
    Node* copyRandomList(Node* head) {
        // 📦 【基礎設施準備】
        // dummy: 配置在 Stack 上的虛擬頭節點，用來穩固地錨定新 List 的起點
        Node dummy(0); 
        Node* prev = &dummy;   // prev: 用來在建構新 List 時，不斷往後串接的尾巴指標
        Node* old_curr = head; // old_curr: 用來走訪原始 List 的探測指標
        
        // 📖 【尋址字典 (Hash Map)】
        // 建立「舊記憶體位址」到「新記憶體位址」的絕對映射
        // Key: 舊節點的指標 (Node*)
        // Value: 對應的新節點的指標 (Node*)
        unordered_map<Node*, Node*> mp;

        // ==========================================
        // 🔄 【Round 1：實體複製與 Next 指標串接】
        // 目標：複製所有節點的 val，串好基礎的 next 鍊，並建檔到字典中
        // ==========================================
        while(old_curr != nullptr){
            
            // 1. 在 Heap 上實體化一個全新的節點，複製原節點的值
            Node* node1 = new Node(old_curr->val);
            
            // 2. 登錄字典：將「舊節點的地址」與「新節點的地址」綁定
            mp[old_curr] = node1;

            // 3. 推進新 List 的 next 指標，將剛做好的新節點接上去
            prev->next = node1;
            prev = prev->next;

            // 4. 原 List 繼續往後走
            old_curr = old_curr->next;
        }

        // ==========================================
        // 🔄 【Round 2：Random 指標的時空對接】
        // 目標：利用字典，將新節點的 random 指向正確的新記憶體位址
        // ==========================================
        
        // 狀態重置：兩個探測指標雙雙回到起點
        old_curr = head;
        Node* new_curr = dummy.next; 

        while(old_curr != nullptr){
            
            // 🛡️ 【防禦性編程：避免 Hash Map 污染】
            if (old_curr->random == nullptr) {
                // 如果原本的 random 指向空，新節點也必須老實地指向空
                new_curr->random = nullptr;
            } else {
                // 透過字典查詢：舊的 random 指向哪個舊節點，新的 random 就指向該舊節點對應的「新節點」
                new_curr->random = mp[old_curr->random];
            }
            /*
             * 💡 為什麼不能無腦寫 new_curr->random = mp[old_curr->random];？
             * 在 C++ 的 unordered_map 中，如果你查詢一個「不存在的 Key (例如 nullptr)」，
             * Map 不會報錯，而是會「自動幫你新建這個 Key」，並賦予 Value 一個預設值 (指標的預設值就是 nullptr)。
             * 雖然結果看起來剛好是 nullptr，但這會在 Map 中殘留一筆 <nullptr, nullptr> 的幽靈垃圾資料。
             * 這種「非預期的副作用 (Side Effect)」在大型系統中是記憶體洩漏與效能拖垮的隱患。
             */

            // 雙線並行推進
            new_curr = new_curr->next;
            old_curr = old_curr->next;
        }
        
        // 回傳新 List 的真正起點
        return dummy.next;
    }
};
