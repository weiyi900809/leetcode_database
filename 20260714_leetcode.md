114. Flatten Binary Tree to Linked List


/**
 * Definition for a binary tree node.
 * struct TreeNode {
 * int val;
 * TreeNode *left;
 * TreeNode *right;
 * TreeNode() : val(0), left(nullptr), right(nullptr) {}
 * TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 * TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    /*
     * 🧠 【核心思維：逆向工程 (Reverse Engineering)】
     * 目標是將樹轉成 Pre-order (中 -> 左 -> 右) 的單向串列。
     * 若正向修改，`node->right` 會被提早覆蓋，導致原本的右子樹遺失。
     * 因此，我們將走訪順序完全反轉，變成「右 -> 左 -> 中 (Reverse Pre-order)」。
     * 這樣我們就能從目標串列的「最後一個節點 (Tail)」開始，由後往前一路建構回「頭部 (Head)」。
     */
    
    // TreeNode*& prev：這是「指標的參考 (Pointer Reference)」。
    // 它讓 prev 變成一個全域共享的變數。在所有遞迴層級中，它永遠指向「目前已經扁平化完成的串列開頭」。
    void helper(TreeNode* node, TreeNode*& prev ){
        
        // 🛡️ 【遞迴終止條件】
        // 踩到空節點，直接返回上一層
        if(node == nullptr) return;
        
        // ==========================================
        // 🔄 【步驟 1 & 2：一路向右，再向左 (顛倒的 DFS)】
        // ==========================================
        // 先潛入最深處，把右子樹徹底扁平化
        helper(node->right, prev);
        
        // 再處理左子樹的扁平化
        helper(node->left, prev);
        
        // ==========================================
        // 🔗 【步驟 3：物理串接 (由後往前)】
        // ==========================================
        // 此時，prev 記錄著「已經在我們身後排好隊的 Linked List 的第一個人」
        
        // 將當前節點的 right 指向這串已經排好的隊伍
        node->right = prev;
        
        // 按照題目要求，將 left 指標清空
        node->left = nullptr;
        
        // ==========================================
        // 📍 【步驟 4：更新進度】
        // ==========================================
        // 當前節點處理完畢，它成為了這條 Linked List 新的「最前端」。
        // 所以把 prev 更新為當前節點，等待上一層的父節點來牽它的手。
        prev = node;
    }
    
    void flatten(TreeNode* root) {
        // prev 初始化為 nullptr，因為串列的最後一個節點的 right 必定指向 nullptr
        TreeNode* prev = nullptr;
        
        // 啟動逆向遞迴
        helper(root, prev);
    }
};
